% hyperref is loaded by pandoc's own LaTeX template, but different pandoc
% versions insert -H/header-includes content either before or after that
% happens. \AtBeginDocument defers this until the whole preamble (including
% hyperref) has been processed, regardless of pandoc's template ordering.
\AtBeginDocument{%
  \hypersetup{colorlinks=true, linkcolor=blue, urlcolor=blue, citecolor=blue}%
  \let\OldHyperref\hyperref
  \renewcommand{\hyperref}[2][]{\OldHyperref[#1]{\ul{#2}}}%
  \let\OldHref\href
  \renewcommand{\href}[2]{\OldHref{#1}{\ul{#2}}}%
}
